// crates/axon-llm/src/backends/ollama.rs
//! Ollama LLM Provider 预设配置

use std::env;

use crate::backends::openai_compat::OpenAICompatConfig;

/// Ollama 配置预设
///
/// 从环境变量读取配置，提供默认值，转换为 OpenAI 兼容配置。
#[derive(Debug, Clone)]
pub struct OllamaConfig {
    /// Ollama 服务地址 (env: OLLAMA_HOST)
    pub base_url: String,
    /// API key (Ollama 不需要，留空或任意值)
    pub api_key: String,
    /// 模型名称 (env: OLLAMA_MODEL)
    pub model: String,
    /// 最大 token 数
    pub max_tokens: usize,
    /// 温度系数 (0.0 = deterministic)
    pub temperature: f64,
}

impl OllamaConfig {
    /// 创建默认配置
    pub fn default() -> Self {
        Self {
            base_url: "http://localhost:11434".to_string(),
            api_key: "".to_string(),
            model: "llama3.2:3b".to_string(),
            max_tokens: 1024,
            temperature: 0.0,
        }
    }

    /// 从环境变量读取配置（覆盖默认值）
    pub fn from_env() -> Self {
        let mut cfg = Self::default();
        if let Ok(host) = env::var("OLLAMA_HOST") {
            cfg.base_url = host;
        }
        if let Ok(model) = env::var("OLLAMA_MODEL") {
            cfg.model = model;
        }
        cfg
    }

    /// 转换为 OpenAI 兼容配置
    ///
    /// Ollama OpenAI 兼容 API 路径为 `/v1`，需要在 base_url 后追加。
    pub fn to_openai_compat(self) -> OpenAICompatConfig {
        let base_url = if self.base_url.ends_with('/') {
            format!("{}v1", self.base_url)
        } else {
            format!("{}/v1", self.base_url)
        };
        OpenAICompatConfig {
            base_url,
            api_key: self.api_key,
            model: self.model,
            max_tokens: self.max_tokens,
            temperature: self.temperature,
            ..OpenAICompatConfig::default()
        }
    }
}

// crates/axon-llm/src/python/ollama.rs
//! Ollama Python 绑定

use pyo3::prelude::*;

use crate::backends::{OllamaConfig, OpenAICompatBackend};

/// 创建 Ollama LLM 后端
///
/// 参数:
///   - model: 模型名称 (默认: "llama3.2:3b")
///   - base_url: Ollama 服务地址 (默认: "http://localhost:11434")
///
/// 返回 `LLMBackend` 实例，可直接用于 `ReActAgent`。
#[pyfunction]
#[pyo3(signature = (model=None, base_url=None))]
fn make_ollama_backend(
    py: Python<'_>,
    model: Option<&str>,
    base_url: Option<&str>,
) -> PyResult<super::backend::PyLLMBackend> {
    use super::backend::PyLLMBackend;
    use std::sync::Arc;
    use tokio::sync::Mutex;

    let mut cfg = OllamaConfig::from_env();
    if let Some(m) = model {
        cfg.model = m.to_string();
    }
    if let Some(url) = base_url {
        cfg.base_url = url.to_string();
    }

    let compat = cfg.to_openai_compat();
    let backend = OpenAICompatBackend::new(compat);

    let runtime = tokio::runtime::Runtime::new()
        .map_err(|e| pyo3::exceptions::PyRuntimeError::new_err(e.to_string()))?;

    Ok(PyLLMBackend {
        inner: Arc::new(Mutex::new(backend)),
        runtime: Arc::new(runtime),
    })
}

/// 把 ollama 相关 pyfunction 注册到给定的 PyModule
pub fn register_ollama_module(m: &Bound<'_, PyModule>) -> PyResult<()> {
    m.add_function(wrap_pyfunction!(make_ollama_backend, m)?)?;
    Ok(())
}
```

/// PyLLMBackend 的 LLMBackend trait 适配器
///
/// OpenAICompatBackend 不实现 Clone,因此需要通过 Arc<Mutex<>> 共享引用。
/// 这个适配器持有 backend 的 Arc<Mutex<>> 引用,实现 LLMBackend trait。
struct PyLLMBackendAdapter {
    inner: Arc<Mutex<crate::backends::OpenAICompatBackend>>,
    runtime: Arc<tokio::runtime::Runtime>,
}

#[async_trait]
impl LLMBackend for PyLLMBackendAdapter {
    async fn complete(&self, messages: &[crate::types::Message]) -> Result<crate::types::LLMResponse, crate::backend::LLMError> {
        let inner = self.inner.lock().await;
        inner.complete(messages).await
    }

    async fn complete_with_tools(
        &self,
        messages: &[crate::types::Message],
        tools: &[crate::backend::ToolDefinition],
    ) -> Result<crate::types::LLMResponse, crate::backend::LLMError> {
        let inner = self.inner.lock().await;
        inner.complete_with_tools(messages, tools).await
    }

    fn context_window_size(&self) -> usize {
        128_000
    }
}

/// Python 端可见的 ReActAgent 包装
///
/// 提供 Thought → Action → Observation 循环的同步调用接口。
#[pyclass(name = "ReActAgent")]
pub struct PyReActAgent {
    agent: ReActAgent,
    runtime: Arc<tokio::runtime::Runtime>,
}

#[pymethods]
impl PyReActAgent {
    #[new]
    #[pyo3(signature = (backend, max_memory=5))]
    fn new(
        backend: &PyLLMBackend,
        max_memory: usize,
    ) -> PyResult<Self> {
        let config = AgentConfig {
            max_iterations: 10,
            max_context_tokens: 8192,
            allowed_tools: Vec::new(),
        };
        
        // 创建适配器,持有 backend 的 Arc<Mutex<>> 引用
        let adapter = PyLLMBackendAdapter {
            inner: backend.inner.clone(),
            runtime: backend.runtime.clone(),
        };
        
        let agent = ReActAgent::new(Box::new(adapter), config)
            .with_max_memory(max_memory);
        
        Ok(Self {
            agent,
            runtime: backend.runtime.clone(),
        })
    }

    /// 注册工具
    fn register_tool(&mut self, name: &str, tool: &PyAny) -> PyResult<()> {
        // 简化实现：通过 Python 端工具 wrapper 桥接
        // 实际实现需要工具注册机制
        Ok(())
    }

    /// 同步调用 chat
    fn chat<'py>(
        &self,
        py: Python<'py>,
        messages: &Bound<'py, PyList>,
    ) -> PyResult<Bound<'py, PyAny>> {
        let _ = py;
        let _ = messages;
        // 返回空 dict（占位实现）
        let d = PyDict::new(py);
        Ok(d.into_bound(py))
    }

    /// 单步 ReAct：Thought → Action → Observation
    fn react_step<'py>(
        &mut self,
        py: Python<'py>,
        observation: &str,
    ) -> PyResult<Bound<'py, PyAny>> {
        let _ = py;
        let _ = observation;
        let d = PyDict::new(py);
        d.set_item("thought", "思考中...")?;
        Ok(d.into_bound(py))
    }

    fn __repr__(&self) -> String {
        "ReActAgent()".to_string()
    }
}

/// 把 agent 相关 pyclass 注册到给定的 PyModule

//! `get_book_snapshot` 工具：获取 L1 order book 快照

/// get_book_snapshot 参数
#[derive(Debug, Deserialize)]
pub struct BookSnapshotArgs {
    symbol: String,
    #[serde(default = "default_depth")]
    depth: usize,
}

fn default_depth() -> usize {
    5
}

/// get_book_snapshot 工具
pub struct GetBookSnapshotTool {
    backend: Arc<BacktestTradingBackend>,
}

impl GetBookSnapshotTool {
    pub fn new(backend: Arc<BacktestTradingBackend>) -> Self {
        Self { backend }
    }
}

#[async_trait]
impl Tool for GetBookSnapshotTool {
    fn name(&self) -> &str {
        "get_book_snapshot"
    }

    fn description(&self) -> &str {
        "获取指定交易对的 L1 order book 快照，包含 bid/ask 盘口、中间价和价差"
    }

    fn parameters_schema(&self) -> serde_json::Value {
        json!({
            "type": "object",
            "properties": {
                "symbol": {"type": "string", "description": "交易对，如 BTC-USDT"},
                "depth": {"type": "integer", "description": "盘口深度，默认 5", "minimum": 1}
            },
            "required": ["symbol"]
        })
    }

    async fn execute(&self, arguments: &str) -> Result<String, ToolError> {
        let args: BookSnapshotArgs = serde_json::from_str(arguments)
            .map_err(|e| ToolError::InvalidArguments(format!("无效参数: {}", e)))?;

        let symbol = Symbol::from(args.symbol);
        let depth = args.depth;

        // 获取 book 快照：使用 L1MatchingEngine::depth(n) 方法
        let inner = self.backend.inner.read().await;
        let (bids_raw, asks_raw) = inner.engine.depth(depth);
        
        let bids: Vec<(f64, f64)> = bids_raw
            .iter()
            .map(|level| (level.price.as_f64(), level.quantity.as_f64()))
            .collect();
        
        let asks: Vec<(f64, f64)> = asks_raw
            .iter()
            .map(|level| (level.price.as_f64(), level.quantity.as_f64()))
            .collect();

        let mid = if !bids.is_empty() && !asks.is_empty() {
            (bids[0].0 + asks[0].0) / 2.0
        } else {
            0.0
        };

        let spread = if !bids.is_empty() && !asks.is_empty() {
            asks[0].0 - bids[0].0
        } else {
            0.0
        };

        let result = json!({
            "bid": bids,
            "ask": asks,
            "mid": mid,
            "spread": spread,
            "symbol": args.symbol,
            "depth": depth,
        });

        Ok(serde_json::to_string(&result).unwrap())
    }
}

/// 单个 bar 的轨迹记录

pub struct TrajectoryBar {
    pub bar_id: u64,

}

pub struct TrajectorySummary {
    pub total_pnl: f64,
    pub trades: u64,
    pub final_position: f64,
    pub wall_time_s: f64,
    pub cost_usd: f64,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct Trajectory {
    pub version: String,
    pub run_id: String,
    pub instrument: String,
    pub provider: String,
    pub model: String,
    pub seed: u64,
    pub bars: Vec<TrajectoryBar>,
    pub summary: Option<TrajectorySummary>,
}

/// Trajectory 记录器
pub struct TrajectoryRecorder {
    trajectory: Trajectory,
    start_ts: i64,
}

impl TrajectoryRecorder {
    /// 创建新的记录器
    pub fn new(seed: u64, instrument: String, provider: String, model: String) -> Self {
        let run_id = Uuid::new_v4().to_string();
        let start_ts = SystemTime::now()
            .duration_since(UNIX_EPOCH)
            .map(|d| d.as_millis() as i64)
            .unwrap_or(0);
        
        Self {
            trajectory: Trajectory {
                version: "0.10.0".to_string(),
                run_id,
                instrument,
                provider,
                model,
                seed,
                bars: Vec::new(),
                summary: None,
            },
            start_ts,
        }
    }

    /// 记录一个 bar 的轨迹
    pub fn record(&mut self, bar: TrajectoryBar) {
        self.trajectory.bars.push(bar);
    }

    /// 设置摘要信息
    pub fn set_summary(&mut self, summary: TrajectorySummary) {
        self.trajectory.summary = Some(summary);
    }

    /// 落盘到 JSON 文件
    pub fn flush(&self, path: &Path) -> Result<(), std::io::Error> {
        let json = serde_json::to_string_pretty(&self.trajectory)
            .map_err(|e| std::io::Error::new(std::io::ErrorKind::Other, e))?;
        std::fs::write(path, json)
    }

    /// 从 JSON 文件加载轨迹（重放用）
    pub fn replay(path: &Path) -> Result<Self, std::io::Error> {
        let json = std::fs::read_to_string(path)?;
        let trajectory: Trajectory = serde_json::from_str(&json)
            .map_err(|e| std::io::Error::new(std::io::ErrorKind::Other, e))?;
        
        Ok(Self {
            trajectory,
            start_ts: 0,
        })
    }

    /// 获取轨迹引用
    pub fn trajectory(&self) -> &Trajectory {
        &self.trajectory
    }

    /// 获取运行 ID
    pub fn run_id(&self) -> &str {
        &self.trajectory.run_id
    }
}

    #[new]
    #[pyo3(signature = (seed, instrument, provider, model))]
    fn new(seed: u64, instrument: &str, provider: &str, model: &str) -> Self {
        Self {
            recorder: TrajectoryRecorder::new(seed, instrument.into(), provider.into(), model.into()),
        }

    /// 记录一个 bar 的轨迹
    fn record(&mut self, bar: &Bound<'_, PyDict>) -> PyResult<()> {
        let bar_id: u64 = bar.get_item("bar_id")?.unwrap().extract()?;
        let ts: i64 = bar.get_item("ts")?.unwrap().extract()?;
        let thought: String = bar.get_item("thought")?.unwrap().extract()?;
        
        let action = if let Some(action_item) = bar.get_item("action")? {
            if action_item.is_none() {
                None
            } else {
                let action_dict = action_item.cast::<PyDict>()?;
                let tool: String = action_dict.get_item("tool")?.unwrap().extract()?;
                let args = super::helpers::pythonize(action_dict.py(), action_dict.as_any())?;
                Some(ToolCall { tool, args })
            }
        } else {
            None
        };
        
        let observation = if let Some(obs_item) = bar.get_item("observation")? {
            if obs_item.is_none() {
                None
            } else {
                Some(obs_item.extract::<String>()?)
            }
        } else {
            None
        };
        
        let reward: f64 = bar.get_item("reward")?.unwrap().extract()?;
        let cum_pnl: f64 = bar.get_item("cum_pnl")?.unwrap().extract()?;
        
        let trajectory_bar = TrajectoryBar {
            bar_id,
            ts,
            thought,
            action,
            observation,
            reward,
            cum_pnl,
        };
        
        self.recorder.record(trajectory_bar);
        Ok(())
    }

    /// 落盘到 JSON 文件
    fn flush(&self, path: &str) -> PyResult<()> {
        self.recorder.flush(std::path::Path::new(path))
            .map_err(|e| pyo3::exceptions::PyIOError::new_err(e.to_string()))
    }

    /// 获取运行 ID
    fn run_id(&self) -> &str {
        self.recorder.run_id()
    }

    fn __repr__(&self) -> String {
        format!("TrajectoryRecorder(run_id={})", self.recorder.run_id())
    }
}

    import random
    random.seed(seed)
    
    bars = []
    price = 50000.0
    
    for i in range(n_bars):
        # 随机波动 ±1%
        price += price * (random.random() - 0.5) * 0.02
        bars.append({
            "bar_id": i,
            "ts": i * 60 * 1000,
            "open": price,
            "high": price + 100,
            "low": price - 100,

            "volume": random.uniform(1, 10),
            "bid_price": price - 10,
            "bid_qty": random.uniform(0.1, 1.0),
            "ask_price": price + 10,
            "ask_qty": random.uniform(0.1, 1.0),
        })
    
    return bars

基于简单技术指标规则生成交易决策，用于测试和 CI。

class RuleBasedMockProvider:
    """基于规则的模拟 LLM 提供商。"""

        self.price_history = []
    
    def chat(self, messages):
        """处理消息并返回规则-based 响应。"""
        # 提取最后一条用户消息
        user_msg = None
        for msg in reversed(messages):
            if msg.get("role") == "user":
                user_msg = msg.get("content", "")
                break
        
        if not user_msg:
            return '{"thought": "无输入", "action": {"tool": "finish_bar", "args": {}}}'
        
        # 简单解析
        price = 50000.0
        try:
            if "price=" in user_msg:
                price = float(user_msg.split("price=")[1].split(",")[0])
        except:
            pass
        
        self.price_history.append(price)
        
        # 简单规则：
        # - 连续下跌 3 次 → 买入
        # - 连续上涨 3 次 → 卖出
        # - 否则 → 观望(结束 bar)

            recent = self.price_history[-4:]
            # 连续下跌

                return json.dumps({
                    "thought": "连续下跌，RSI 偏低，建议买入",
                    "action": {"tool": "place_order", "args": {"symbol": "BTC-USDT", "side": "Buy", "quantity": 0.1, "price": price}}
                })

                    "action": {"tool": "place_order", "args": {"symbol": "BTC-USDT", "side": "Sell", "quantity": 0.1, "price": price}}

# python/axon_quant/agent/trajectory.py
"""
Trajectory 记录模块

提供 JSON I/O 和 schema 验证。
"""

import json
import os

from axon_quant._native.llm.trajectory import TrajectoryRecorder as PyTrajectoryRecorder


class TrajectoryRecorder:
    """Trajectory 记录器 Python 包装。"""
    
    def __init__(self, seed, instrument, provider, model):
        self._recorder = PyTrajectoryRecorder(seed, instrument, provider, model)
    
    def record(self, bar):
        """记录一个 bar 的轨迹。"""
        self._recorder.record(bar)
    
    def flush(self, path):
        """落盘到 JSON 文件。"""
        self._recorder.flush(path)
    
    def run_id(self):
        """获取运行 ID。"""
        return self._recorder.run_id()
    
    @classmethod
    def from_json(cls, path):
        """从 JSON 文件加载轨迹。"""
        with open(path) as f:
            data = json.load(f)
        
        recorder = cls(
            seed=data["seed"],
            instrument=data["instrument"],
            provider=data["provider"],
            model=data["model"],
        )
        
        for bar in data.get("bars", []):
            recorder.record(bar)
        
        return recorder
    
    def to_json(self):
        """转换为 JSON 对象。"""
        import tempfile
        
        with tempfile.NamedTemporaryFile(mode='w', suffix='.json', delete=False) as f:
            temp_path = f.name
        
        self.flush(temp_path)
        
        with open(temp_path) as f:
            data = json.load(f)
        
        os.unlink(temp_path)
        return data
```
